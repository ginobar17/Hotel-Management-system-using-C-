class Verification {
protected:
	bool aadhaar;

public:
	void verify() {
		cout << "Do you have an Aadhaar card? (1: Yes, 0: No): ";
		cin >> aadhaar;
	}

	bool isIndian() const {
		return aadhaar;
	}
};

class Customer : public Person, public Verification {
protected:
	int customer_id, age;
	char address[50];

public:
	void customerc() {
		details();
		cout << "Enter your age: ";
		while (!(cin >> age) || age <= 0) {
			cout << "Please enter a valid age: ";
			cin.clear();
			cin.ignore(1000, '\n'); // Ignore until the next newline
		}
		cin.ignore(); // Clear the newline character from the buffer
		cout << "Enter your address: ";
		cin.getline(address, sizeof(address));
		customer_id = rand() % 100 + 1;

		// Verify nationality
		verify();
	}

	double calculateBill(int days) {
		const int room_rate = 100; // Room rate per day
		double total_bill = room_rate * days;
		double discount = 0.0;

		// Apply discount for Indian customers
		if (isIndian()) {
			discount = total_bill * 0.1; // 10% discount
			total_bill -= discount;
			cout << "Original Price: $" << room_rate * days << endl;
			cout << "Discount applied: 10% ($" << discount << ")\n";
		}

		return total_bill;
	}
};

class Room {
protected:
	bool room_occupied[100] = { false };

public:

	int checkAvailability() {
		for (int i = 0; i < 100; i++) {
			if (!room_occupied[i]) {
				cout << "Room " << (i + 1) << " is available." << endl;
				return i + 1; // Room numbers are 1-100
			}
		}
		cout << "No rooms available." << endl;
		return -1;
	}

	int checkroomAvailability(int r) {
		int num, flag1 = -1;
		ifstream file("data.txt", ios::beg);
		while (file >> num)
		{
			if (num == r)
			{
				return r + 1; // Room numbers are 1-100
				flag1 = 1;
			}
		}

		if (flag1 == -1)
			return -1;
	}

	void checkin(int room_no) {
		fstream file("data.txt",ios::app);
		if (room_no > 0 && room_no <= 100 && room_occupied[room_no - 1] != true) {
			room_occupied[room_no - 1] = true;
			total_no_of_rooms_occupied++;
			file << room_no << "\n";

		}
	}

	void checkout(int room_no) {
		int num;
		if (room_no > 0 && room_no <= 100 && room_occupied[room_no - 1]) {
			room_occupied[room_no - 1] = false;
			total_no_of_rooms_occupied--;
			
		}
	}
};
